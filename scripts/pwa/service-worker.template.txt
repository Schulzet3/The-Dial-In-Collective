// Build-injected fingerprint follows the exact revision of these public assets.
// Strict allowlist: no fetch interception for Supabase, RPC, admin, or member data.
const CACHE_NAME='dic-public-shell-__DIC_BUILD_ID__';
const SHELL=__DIC_ASSETS__;
const ALLOWED=new Set(SHELL);
const NAVIGATION=new Set(['/','/index.html']);
self.addEventListener('install',event=>{
 // Precache the *complete* revision before activating; a perpetually open
 // iOS Home Screen app must not keep the old UI pinned indefinitely.
 // Do not force-reload an editing/logging tab: the next navigation upgrades.
 event.waitUntil(caches.open(CACHE_NAME).then(cache=>cache.addAll(SHELL))
  .then(()=>self.skipWaiting()));
});
self.addEventListener('activate',event=>{
 event.waitUntil(caches.keys().then(names=>Promise.all(names
  .filter(name=>name.startsWith('dic-public-shell-')&&name!==CACHE_NAME)
  .map(name=>caches.delete(name)))));
 // No clients.claim(): never change control of an already active tab.
});
self.addEventListener('fetch',event=>{
 const request=event.request;
 if(request.method!=='GET')return;
 const url=new URL(request.url);
 if(url.origin!==self.location.origin||url.search||!ALLOWED.has(url.pathname))return;
 if(request.mode==='navigate'){
  if(!NAVIGATION.has(url.pathname))return;
  // Online startup checks the live HTML, so a cached install cannot mask a
  // published release. Offline startup still uses the same cached shell.
  event.respondWith(fetch(request,{cache:'no-store'}).then(response=>{
   if(response.ok)return response;
   throw new Error('shell_navigation_unavailable');
  }).catch(async()=>{
   const cache=await caches.open(CACHE_NAME);
   return await cache.match(url.pathname)||await cache.match('/index.html');
  }));
  return;
 }
 event.respondWith(caches.open(CACHE_NAME).then(async cache=>
  await cache.match(url.pathname)||fetch(request)));
});
