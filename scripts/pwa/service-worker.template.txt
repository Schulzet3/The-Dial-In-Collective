// Build-injected fingerprint follows the exact revision of these public assets.
// Strict allowlist: no fetch interception for Supabase, RPC, admin, or member data.
const CACHE_NAME='dic-public-shell-__DIC_BUILD_ID__';
const SHELL=__DIC_ASSETS__;
const ALLOWED=new Set(SHELL);
const NAVIGATION=new Set(['/','/index.html']);
self.addEventListener('install',event=>{
 // Precache the *complete* revision before activating; a perpetually open
 // iOS Home Screen app must not keep the old UI pinned indefinitely.
 // Do not force-reload an editing/logging tab: the next navigation upgrades.
 event.waitUntil(caches.open(CACHE_NAME).then(cache=>cache.addAll(SHELL))
  .then(()=>self.skipWaiting()));
});
// After skipWaiting, existing tabs still use their previous worker and cache.
// Deleting that cache on activation breaks an offline reload of those tabs.
// Retire old public shells only when every open window is controlled by this
// worker, then retry cheaply on a later static fetch. Never touch member data.
const SHELL_PREFIX='dic-public-shell-';
let shellCleanupDone=false,shellCleanupInFlight=null,nextShellCleanupAt=0;
async function pruneUnusedShells(){
 const [all,controlled]=await Promise.all([
  self.clients.matchAll({type:'window',includeUncontrolled:true}),
  self.clients.matchAll({type:'window'})
 ]);
 const currentIds=new Set(controlled.map(client=>client.id));
 if(all.some(client=>!currentIds.has(client.id)))return false;
 const names=await caches.keys();
 await Promise.all(names.filter(name=>name.startsWith(SHELL_PREFIX)&&name!==CACHE_NAME)
  .map(name=>caches.delete(name)));
 return true;
}
function scheduleShellCleanup(event){
 if(shellCleanupDone||shellCleanupInFlight||Date.now()<nextShellCleanupAt)return;
 nextShellCleanupAt=Date.now()+12000;
 shellCleanupInFlight=pruneUnusedShells()
  .then(done=>{shellCleanupDone=done;})
  .catch(()=>{/* Leave old shell cached until another safe opportunity. */})
  .finally(()=>{shellCleanupInFlight=null;});
 event.waitUntil(shellCleanupInFlight);
}
self.addEventListener('activate',event=>{
 scheduleShellCleanup(event);
 // No clients.claim(): never change control of an already active tab.
});
self.addEventListener('fetch',event=>{
 const request=event.request;
 if(request.method!=='GET')return;
 const url=new URL(request.url);
 if(url.origin!==self.location.origin||url.search||!ALLOWED.has(url.pathname))return;
 scheduleShellCleanup(event);
 if(request.mode==='navigate'){
  if(!NAVIGATION.has(url.pathname))return;
  // Online startup checks the live HTML, so a cached install cannot mask a
  // published release. Offline startup still uses the same cached shell.
  event.respondWith(fetch(request,{cache:'no-store'}).then(response=>{
   if(response.ok)return response;
   throw new Error('shell_navigation_unavailable');
  }).catch(async()=>{
   const cache=await caches.open(CACHE_NAME);
   return await cache.match(url.pathname)||await cache.match('/index.html');
  }));
  return;
 }
 event.respondWith(caches.open(CACHE_NAME).then(async cache=>
  await cache.match(url.pathname)||fetch(request)));
});


// iOS 16.4+ Home Screen Web Push. Server sends no user/member details and no
// encrypted payload. Never cache identity, tokens, notifications, or records.
self.addEventListener('push',event=>{
 event.waitUntil(self.registration.showNotification('The Dial In Collective',{
  body:'Administrator review required.',
  tag:'dic-admin-review',
  icon:'/icon-192.png',
  data:{path:'/admin.html'}
 }));
});
self.addEventListener('notificationclick',event=>{
 event.notification.close();
 event.waitUntil((async()=>{
  const windows=await self.clients.matchAll({type:'window',includeUncontrolled:true});
  const existing=windows.find(client=>{
   try{return new URL(client.url).origin===self.location.origin &&
    new URL(client.url).pathname==='/admin.html';}catch{return false;}
  });
  if(existing)return existing.focus();
  // Do not navigate an open shot draft to Admin, which could lose local edits.
  return self.clients.openWindow('/admin.html');
 })());
});
